import json
import os
import socket
import struct
import subprocess
import sys
import tempfile
import threading
from pathlib import Path

repo = Path('/mnt/d/PEX-work')
sys.path.insert(0, str(repo))
from benchmarks.linux_sandbox import worker_relay_command
from benchmarks.worker_model_proxy import _read

with tempfile.TemporaryDirectory(prefix='pex-worker-proxy-') as temp:
    root = Path(temp)
    workspace = root/'worker'
    workspace.mkdir()
    controller = root/'controller'
    controller.mkdir(mode=0o700)
    path = controller/'relay.sock'
    listener = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
    listener.bind(str(path))
    path.chmod(0o600)
    listener.listen(1)
    listener.settimeout(20)
    calls=[]
    errors=[]
    def serve():
        try:
            with listener.accept()[0] as peer:
                peer.settimeout(10)
                length=struct.unpack('!I',_read(peer,4))[0]
                assert 0<length<=262144
                envelope=json.loads(_read(peer,length))
                body=envelope['body']
                assert body['model']=='pinned' and body['stream'] is False
                calls.append(body)
                response={'schema':envelope['schema'],'request_id':envelope['request_id'],'ok':True,
                    'body':{'id':'controlled','created':1,'model':'pinned','choices':[
                        {'index':0,'message':{'role':'assistant','content':'bounded response'},'finish_reason':'stop'}],
                        'usage':{'prompt_tokens':10,'completion_tokens':2,'total_tokens':12}}}
                raw=json.dumps(response).encode()
                peer.sendall(struct.pack('!I',len(raw))+raw)
        except Exception as exc:
            errors.append(type(exc).__name__)
    thread=threading.Thread(target=serve,daemon=True)
    thread.start()
    source=(repo/'benchmarks/worker_model_proxy.py').read_text()
    child="""
import http.client,json,threading
namespace={'__name__':'proxy_module'}
exec(SOURCE,namespace)
server=namespace['make_server']('/model-relay.sock',model='pinned')
thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()
try:
    client=http.client.HTTPConnection('127.0.0.1',server.server_port,timeout=10)
    client.request('POST','/v1/chat/completions',body=json.dumps({'model':'pinned','messages':[{'role':'user','content':'ping'}],'max_tokens':20,'stream':True}))
    response=client.getresponse();raw=response.read();client.close()
    assert response.status==200 and b'bounded response' in raw and raw.endswith(b'data: [DONE]\\n\\n')
    print(json.dumps({'loopback_http':True,'mounted_unix_relay':True,'buffered_sse':True,'upstream_streaming':False}))
finally:
    server.shutdown();server.server_close();thread.join(timeout=5)
""".replace('SOURCE',repr(source))
    try:
        result=subprocess.run(worker_relay_command(workspace,['/usr/bin/python3','-c',child],path),
            capture_output=True,text=True,timeout=30)
        thread.join(timeout=2)
        assert result.returncode==0, result.stderr
        assert len(calls)==1 and not errors and not thread.is_alive(),errors
        receipt={**json.loads(result.stdout),'controller_calls':len(calls),
            'scope':'Controlled IPC transport inside Linux bubblewrap; no coding harness or upstream provider call.'}
        (repo/'build/worker-proxy-linux-smoke.json').write_text(json.dumps(receipt,indent=2)+'\n')
        print(json.dumps(receipt))
    finally:
        listener.close()
