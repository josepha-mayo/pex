import asyncio
import hashlib
import json
import time
import os
import socket
import struct
import subprocess
import sys
import tempfile
import threading
from pathlib import Path

repo = Path('/mnt/d/PEX-work')
sys.path[:0]=[str(repo),str(repo/'build/isolated-supervisor-runtime/public-runtime-f379031/site-packages')]
from benchmarks.linux_sandbox import worker_runtime_relay_command
from benchmarks.worker_model_proxy import _read
from benchmarks.opencode_control_relay import OpenCodeControlRelay
from benchmarks.pex_attach import _observe_controlled_workspace, _decide_out_of_process
from pex_protocol.session import HarnessSession
from pex_protocol.enums import HarnessType,SessionStatus

with tempfile.TemporaryDirectory(prefix='pex-worker-proxy-') as temp:
    root = Path(temp)
    workspace = root/'worker'
    workspace.mkdir()
    task='Fix solver.py and verify test_public.py.'
    (workspace/'solver.py').write_text('def answer():\n    return 0\n')
    (workspace/'TASK.md').write_text(task)
    public=workspace/'test_public.py'
    public.write_text("import importlib\nsolver = importlib.import_module('solver')\ndef test_answer():\n    assert solver.answer() == 1\n")
    fingerprint=hashlib.sha256(public.read_bytes()).hexdigest()
    decisions=[]
    phase=[0]
    runtime=root/'runtime'
    runtime.mkdir()
    import tarfile
    package=repo/'build/linux-opencode-runtime/opencode-linux-x64-1.18.32.tgz'
    with tarfile.open(package) as archive:
        member=archive.getmember('package/bin/opencode')
        assert member.isfile() and member.size<200_000_000
        with archive.extractfile(member) as source, (runtime/'opencode').open('wb') as target:
            import shutil
            shutil.copyfileobj(source,target)
    (runtime/'opencode').chmod(0o755)
    (runtime/'guard.txt').write_text('original')
    (runtime/'benchmarks').mkdir()
    (runtime/'benchmarks/__init__.py').write_text('')
    for name in ('async_budget.py','opencode_cli.py','opencode_worker.py','opencode_session.py','opencode_review_client.py','worker_model_proxy.py'):
        (runtime/'benchmarks'/name).write_bytes((repo/'benchmarks'/name).read_bytes())
    controller = root/'controller'
    controller.mkdir(mode=0o700)
    path = controller/'relay.sock'
    calls=[]
    tool_issued=[]
    errors=[]
    deadline=time.perf_counter()+45
    async def review(vendor,messages):
        session=HarnessSession(id='opencode:'+vendor,harness_type=HarnessType.OPENCODE,vendor_session_id=vendor,cwd=str(workspace),project_id=str(workspace),status=SessionStatus.STOPPED)
        observation=_observe_controlled_workspace(workspace,fingerprint,deadline=deadline,isolated_tests=True)
        decision=await _decide_out_of_process(task_md=task,workspace=workspace,session=session,observed=observation,agent_messages=list(messages),goal_id='public-'+session.id,control_dir=controller/'reviews',timeout=min(15,deadline-time.perf_counter()),offline_runtime=repo/'build/isolated-supervisor-runtime/public-runtime-f379031')
        decisions.append({'decision':decision,'pytest_ok':observation['pytest']['ok'],'controller_verification':observation['controller_verification']})
        if len(decisions)==1:
            assert not observation['pytest']['ok'] and decision['action']['type']=='SEND_NUDGE'
            phase[0]=1
        else:
            assert len(decisions)==2 and observation['pytest']['ok'] and decision['action']['type']=='NOOP'
        return decision['action']
    async def backend(body):
        calls.append(body)
        tools=body.get('tools') or []
        write_tool=next((row.get('function') for row in tools if row.get('function',{}).get('name')=='write'),None)
        message={'role':'assistant','content':'All done.'}
        finish='stop'
        if write_tool and len(tool_issued)<=phase[0]:
            tool_issued.append(write_tool)
            message={'role':'assistant','content':None,'tool_calls':[{'id':'controlled-write-'+str(len(tool_issued)),'type':'function','function':{'name':'write','arguments':json.dumps({'filePath':'/workspace/solver.py','content':'def answer():\n    return '+str(phase[0])+'\n'})}}]}
            finish='tool_calls'
        return {'id':'controlled','created':1,'model':'pinned','choices':[{'index':0,'message':message,'finish_reason':finish}],'usage':{'prompt_tokens':10,'completion_tokens':2,'total_tokens':12}}
    relay=OpenCodeControlRelay(model='pinned',max_calls=8,deadline=deadline,backend=backend,review=review,max_reviews=3)
    original_dispatch = relay.dispatch
    async def audited_dispatch(raw):
        response = await original_dispatch(raw)
        if response.get('ok') is not True:
            errors.append(response.get('error'))
        return response
    relay.dispatch = audited_dispatch
    loop = asyncio.new_event_loop()
    thread = threading.Thread(target=loop.run_forever, daemon=True)
    thread.start()
    try:
        listener_server = asyncio.run_coroutine_threadsafe(relay.listen(path), loop).result(5)
    except BaseException:
        loop.call_soon_threadsafe(loop.stop)
        thread.join(2)
        if not thread.is_alive():
            loop.close()
        raise
    def shutdown_relay():
        if loop.is_running():
            async def shutdown():
                listener_server.close()
                await listener_server.wait_closed()
                await relay.close_active()
            try:
                asyncio.run_coroutine_threadsafe(shutdown(), loop).result(5)
            finally:
                loop.call_soon_threadsafe(loop.stop)
                thread.join(2)
            assert not thread.is_alive() and not relay._handlers
            loop.close()
    source=(repo/'benchmarks/worker_model_proxy.py').read_text()
    child="\nimport asyncio,json,time,sys\nfrom pathlib import Path\nsys.path.insert(0,'/worker-runtime')\nfrom benchmarks.opencode_worker import run_worker\ndeadline=DEADLINE\nlogs=Path('/tmp/turn-logs')\nresult=asyncio.run(run_worker(executable=Path('/worker-runtime/opencode'),workspace=Path('/workspace'),socket_path='/model-relay.sock',model='pinned',prompt='Fix solver.py and verify test_public.py.',log_directory=logs,state_home=Path('/tmp/worker-state'),deadline=deadline,supervised=True,max_followups=2))\nassert len(result.turns)==2 and result.actions==('SEND_NUDGE','NOOP')\nassert result.turns[0].session_id==result.turns[1].session_id\nassert not result.followup_limit_reached and len(result.outgoing_messages)==1\nraw=b''.join((logs/f'turn-{index}.jsonl').read_bytes() for index in range(2))\nevents=[json.loads(line) for line in raw.splitlines()]\nwrites=[event for event in events if event.get('type')=='tool_use' and event.get('part',{}).get('tool')=='write']\nassert len(writes)==2 and all(row['part']['state']['status']=='completed' for row in writes)\nPath('/workspace/_proof-stdout.jsonl').write_bytes(raw)\nprint(json.dumps({'shared_session_deadline':True,'real_opencode':True,'loopback_http':True,'mounted_unix_relay':True,'reusable_worker_entry':True,'upstream_streaming':False,'task_writable':True,'same_vendor_session':True,'pex_followups':1,'public_tests_before':False,'public_tests_after':True,'worker_exit':0}))\n".replace('DEADLINE',repr(deadline))
    try:
        result=subprocess.run(worker_runtime_relay_command(workspace,['/usr/bin/python3','-c',child],path,runtime),
            capture_output=True,text=True,timeout=60)
        shutdown_relay()
        (repo/'build/opencode-worker-linux-diagnostic.json').write_text(json.dumps({'returncode':result.returncode,'errors':errors,'calls':len(calls),'stderr':result.stderr},indent=2))
        assert result.returncode==0, result.stderr
        assert len(tool_issued)==2 and (workspace/'solver.py').read_bytes()==b'def answer():\n    return 1\n'
        assert len(decisions)==2 and not errors
        (repo/'build/opencode-worker-linux-audit.json').write_text(json.dumps({'model':relay.audit,'review':relay.review_audit},indent=2))
        (repo/'build/opencode-worker-linux-decisions.json').write_text(json.dumps(decisions,indent=2))
        assert 1<=len(calls)<=8 and not errors and not thread.is_alive(),errors
        (repo/'build/opencode-worker-linux-stdout.jsonl').write_bytes((workspace/'_proof-stdout.jsonl').read_bytes())
        receipt={**json.loads(result.stdout),'controller_calls':len(calls),'reusable_async_listener':True,'relay_handlers_settled':not relay._handlers,
            'scope':'Real OpenCode same-session repair with isolated deterministic PEX and public tests, controlled relay backend; no upstream provider call or comparative quality claim.'}
        (repo/'build/opencode-worker-linux-smoke.json').write_text(json.dumps(receipt,indent=2)+'\n')
        print(json.dumps(receipt))
    finally:
        shutdown_relay()
